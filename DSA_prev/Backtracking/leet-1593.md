class Solution {
public:
    unordered_set<string> st;
    int maxCount = 0;

    void solve(string &s,int idx,int count) {
        if(count + (s.length() - idx) <= maxCount) return; // pruning

        if(idx == s.length()) maxCount = max(maxCount, count);

        for(int i=idx; i<s.length();i++) {
            string sub = s.substr(idx, i-idx+1);
            if(st.find(sub) == st.end()) {
                st.insert(sub);
                solve(s, i+1, count+1);
                st.erase(sub);
            }
        }
    }

    int maxUniqueSplit(string s) {
        int currCount = 0;
        solve(s, 0,currCount);

        return maxCount;
    }
};
