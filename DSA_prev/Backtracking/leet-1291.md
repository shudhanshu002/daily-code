## method 1 without backtracking

```cpp
class Solution {
public:
    vector<int> sequentialDigits(int low, int high) {
        vector<int> ans;
        for(int i=1;i<9;i++) {
            int x = i;
            for(int nxt = i+1;nxt<=9;nxt++) {
                x = x*10 + nxt;
                if(low <= x && x <= high) ans.push_back(x);
            }
        }

        sort(ans.begin(), ans.end());
        return ans;
    }
};
```

## with backtracking

```cpp
class Solution {
public:
    vector<int> result;

    void backtrack(int low, int high, int start) {
        if(start > high) return;

        if(start >= low && start <= high) result.push_back(start);

        int lastDigit = start % 10;
        if(lastDigit < 9) {
            int nextNum = start * 10 + (lastDigit + 1);
            backtrack(low, high, nextNum);
        }
    } 

    vector<int> sequentialDigits(int low, int high) {
        for(int i=1;i<9;i++) {
            backtrack(low,high,i);
        }
        sort(result.begin(), result.end());
        return result;
    }
};
```
